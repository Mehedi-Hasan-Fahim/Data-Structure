#include<iostream>
using namespace std;

typedef struct node Node;

// Structure for a node
struct node {
    int data;
    Node* left;
    Node* right;
};

// Function to create a new node
Node* createNode(int data) {
    Node* newNode =new Node ;
    newNode->data = data;
    newNode->left = NULL;
    newNode->right = NULL;
    return newNode;
}

// Function to insert a node in the binary tree
Node* insertNode(Node* root, int data) {
    if (root == NULL) {
        root = createNode(data);
        return root;
    }
    else {
        char choice;
        cout<<"Enter a to insert left of "<<root->data<<" or b to insert right of "<<root->data;
        cin>>choice;

        switch (choice) {
        case 'a':
            root->left = insertNode(root->left, data);
            break;
        case 'b':
            root->right = insertNode(root->right, data);
            break;
        default:
            cout<<"Invalid choice! Please enter a or b.\n";
            break;
        }
        return root;
    }
}

// Function to perform preorder traversal of the binary tree
void preorderTraversal(Node* root) {
    if (root == NULL){
        return;
    }
    cout<<root->data<<" ";
    if(root->left!=NULL){
        preorderTraversal(root->left);
    }
    if(root->right!=NULL){
        preorderTraversal(root->right);
    }
}

// Function to perform inorder traversal of the binary tree
void inorderTraversal(Node* root) {
   if (root == NULL){
        return;
    }
    if(root->left!=NULL){
       inorderTraversal(root->left);
    }
    cout<<root->data<<" ";
    if(root->left!=NULL){
       inorderTraversal(root->right);
    }
}

// Function to perform postorder traversal of the binary tree
void postorderTraversal(Node* root) {
    if (root == NULL){
        return;
    }
    if(root->left!=NULL){
       postorderTraversal(root->left);
    }
    if(root->left!=NULL){
       postorderTraversal(root->right);
    }
     cout<<root->data<<" ";
}

// Function to display the binary tree based on traversal option
void displayBinaryTree(Node* root) {
    int option;
    cout<<"\nSelect traversal option:\n";
    cout<<"1. Preorder Traversal\n";
    cout<<"2. Inorder Traversal\n";
    cout<<"3. Postorder Traversal\n";
    cout<<"Enter your choice: ";
    cin>>option;

    switch (option) {
    case 1:
        cout<<"\nPreorder traversal: ";
        preorderTraversal(root);
        break;
    case 2:
        cout<<"\nInorder traversal: ";
        inorderTraversal(root);
        break;
    case 3:
        cout<<"\nPostorder traversal: ";
        postorderTraversal(root);
        break;
    default:
        cout<<"Invalid choice! Please enter a valid option.\n";
        break;
    }

    printf("\n");
}

int main() {
    Node* root = NULL;
    int choice, data;

    while (1) {
        cout<<"\nBinary Tree Operations\n";
        cout<<"1. Insert Node\n";
        cout<<"2. Display Binary Tree\n";
        cout<<"3. Exit\n";
        cout<<"Enter your choice: ";
        cin>>choice;

        switch (choice) {
        case 1:
            cout<<"Enter data to insert: ";
            cin>>data;
            root = insertNode(root, data);
            break;
        case 2:
            displayBinaryTree(root);
            break;
        case 3:
            cout<<"Exiting...\n";
            exit(0);
        default:
            cout<<"Invalid choice! Please enter a valid option.\n";
            break;
        }
    }

    return 0;
}
