#include <iostream>
using namespace std;

void displayArray(int a[], int size) {
    cout << "Array elements: ";
    for (int i = 0; i < size; i++) {
        cout << a[i] << " ";
    }
}

void insertElement(int a[], int &size) {
    int index, number;

    cout << "Insertion index:";
    cin >> index;

    if (index < 0 || index > size + 1) {
        cout << "Invalid index\n";
    } else {
        cout << "Enter insertion number:";
        cin >> number;

        for (int i = size - 1; i >= index; i--) {
            a[i + 1] = a[i];
        }

        a[index] = number;
        size++;
    }
}

void deleteElement(int a[], int &size) {
    int index;

    cout << "Deletion index:";
    cin >> index;

    if (index < 0 || index >= size) {
        cout << "Invalid index\n";
    } else {
        for (int i = index; i < size - 1; i++) {
            a[i] = a[i + 1];
        }
        size--;
    }
}

int main() {
    int a[50], size;
    char choice;

    cout << "Enter Input Size:";
    cin >> size;

    cout << "Enter Array Elements:" << endl;
    for (int i = 0; i < size; i++) {
        cin >> a[i];
    }

    while (true) {
        cout << "\nMenu:\n";
        cout << "1. Insertion\n";
        cout << "2. Deletion\n";
        cout << "3. Show Array\n";
        cout << "4. Exit\n";
        cout << "Enter your choice: ";
        cin >> choice;

        switch (choice) {
            case '1':
                insertElement(a, size);
                break;

            case '2':
                deleteElement(a, size);
                break;

            case '3':
                displayArray(a, size);
                break;

            case '4':
               exit(0);
            default:
                cout << "Invalid choice. Please enter a valid option.\n";
        }
    }

    return 0;
}
