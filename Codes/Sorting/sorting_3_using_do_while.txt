void show(int A[], int size) {
    for (int i = 0; i < size; i++) {
        cout << A[i] << " ";
    }
    cout << endl;
}

void asc_insertion_sort(int A[], int size) {
    int i, j, temp;
    for (i = 1; i < size; i++) {
        temp = A[i];
        j = i - 1;

        while (j >= 0 && A[j] > temp) {
            A[j + 1] = A[j];
            j--;
        }
        A[j + 1] = temp;
    }
}

void dsc_insertion_sort(int A[], int size) {
    int i, j, temp;
    for (i = 1; i < size; i++) {
        temp = A[i];
        j = i - 1;

        while (j >= 0 && A[j] < temp) {
            A[j + 1] = A[j];
            j--;
        }
        A[j + 1] = temp;
    }
}

void asc_selection_sort(int A[], int size) {
    int temp, i, j, index_min;
    for (i = 0; i < size - 1; i++) {
        index_min = i;
        for (j = i + 1; j < size; j++) {
            if (A[j] < A[index_min]) {
                index_min = j;
            }
        }
        if (index_min != i) {
            temp = A[i];
            A[i] = A[index_min];
            A[index_min] = temp;
        }
    }
}

void dsc_selection_sort(int A[], int size) {
    int temp, i, j, index_max;
    for (i = 0; i < size - 1; i++) {
        index_max = i;
        for (j = i + 1; j < size; j++) {
            if (A[j] > A[index_max]) {
                index_max = j;
            }
        }
        if (index_max != i) {
            temp = A[i];
            A[i] = A[index_max];
            A[index_max] = temp;
        }
    }
}

void asc_bubble_sort(int A[], int size) {
    int i, j, temp;
    for (i = 0; i < size - 1; i++) {
        for (j = 0; j < size - i - 1; j++) {
            if (A[j] > A[j + 1]) {
                temp = A[j];
                A[j] = A[j + 1];
                A[j + 1] = temp;
            }
        }
    }
}

void dsc_bubble_sort(int A[], int size) {
    int i, j, temp;
    for (i = 0; i < size - 1; i++) {
        for (j = 0; j < size - i - 1; j++) {
            if (A[j] < A[j + 1]) {
                temp = A[j];
                A[j] = A[j + 1];
                A[j + 1] = temp;
            }
        }
    }
}

void asc_merge_sort(int A[],int left,int mid,int right){
    
    
    int n1=mid-left+1;//size_left
    int n2=right-mid;//size_right
    
    int L[n1],R[n2];
    for(int i=0;i<n1;i++){
        L[i]=A[left+i];
    }
    for(int i=0;i<n2;i++){
        R[i]=A[mid+1+i];
    }
    
    int i,j,k;
    i=0;j=0;k=left;
    
    while(i<n1 && j<n2){
        if(L[i]<=R[j]){
           A[k]=L[i];
           i++;
        }else{
            A[k]=R[j];
            j++;
        }
        k++;
    }
    
    while (i < n1) {
        A[k] = L[i];
        i++;
        k++;
    }
    
    while (j < n2) {
        A[k] = R[j];
        j++;
        k++;
    }
    
}
void dsc_merge_sort(int A[],int left,int mid,int right){
    
    
    int n1=mid-left+1;//size_left
    int n2=right-mid;//size_right
    
    int L[n1],R[n2];
    for(int i=0;i<n1;i++){
        L[i]=A[left+i];
    }
    for(int i=0;i<n2;i++){
        R[i]=A[mid+1+i];
    }
    
    int i,j,k;
    i=0;j=0;k=left;
    
    while(i<n1 && j<n2){
        if(L[i]>=R[j]){
           A[k]=L[i];
           i++;
        }else{
            A[k]=R[j];
            j++;
        }
        k++;
    }
    
    while (i < n1) {
        A[k] = L[i];
        i++;
        k++;
    }
    
    while (j < n2) {
        A[k] = R[j];
        j++;
        k++;
    }
    
}


void asc_merge_divide(int A[],int left,int right){
    
    if(left>=right){
        return ;
    }
    int mid =left +(right-left)/2;
    asc_merge_divide(A,left,mid);
    asc_merge_divide(A,mid+1,right);
    asc_merge_sort(A,left,mid,right);
}
void dsc_merge_divide(int A[],int left,int right){
    
    if(left>=right){
        return ;
    }
    int mid =left +(right-left)/2;
    dsc_merge_divide(A,left,mid);
    dsc_merge_divide(A,mid+1,right);
    dsc_merge_sort(A,left,mid,right);
}


void swap(int &a, int &b) {
  int t = a;
  a = b;
  b = t;
}

int asc_partition(int A[],int l,int h){
    int pivot=A[l];
    int i=l;int j=h;
    while(i<j){
        while(A[i]<=pivot){
            i++;
        }
        while(A[j]>pivot){
            j--;
        }
       if(i<j){ 
        swap(A[i],A[j]);
       }
    }
    swap(A[l],A[j]);
    return j;
}


void asc_quick_sort(int A[],int l,int h){
    if(l<h){
        int j=asc_partition(A,l,h);
        asc_quick_sort(A,l,j-1);
        asc_quick_sort(A,j+1,h);
    }
}


int dsc_partition(int A[],int l,int h){
    int pivot=A[l];
    int i=l;int j=h;
    while(i<j){
        while(A[i]>=pivot){
            i++;
        }
        while(A[j]<pivot){
            j--;
        }
       if(i<j){ 
        swap(A[i],A[j]);
       }
    }
    swap(A[l],A[j]);
    return j;
}


void dsc_quick_sort(int A[],int l,int h){
    if(l<h){
        int j=dsc_partition(A,l,h);
        dsc_quick_sort(A,l,j-1);
        dsc_quick_sort(A,j+1,h);
    }
}
