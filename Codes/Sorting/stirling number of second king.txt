#include<iostream>
using namespace std;

int stirling_num_second_kind(int n,int k){
    if(n == 0 && k == 0 || k > n) return 0;
    if(n == k || k == 1) return 1;
    
    return  stirling_num_second_kind(n-1,k-1) + k*stirling_num_second_kind(n-1,k);
}

int main(){
    /*
    𝑆(𝑛,𝑘) এর মানে হল আমাদের কাছে nটি জিনিস আছে  এবং আমরা এগুলোকে k ভাগে ভাগ করতে চাই , 
   𝑆(𝑛,𝑘) এর মানই হল কত ভাবে এই ভাগ করা যাবে।
    */
    int n,k;
    cout<<"Enter the value of n: ";
    cin>>n;
    cout<<"Enter the value of k: ";
    cin>>k;
    cout<<"S("<<n<<","<<k<<") = "<<stirling_num_second_kind( n, k);
    
}
