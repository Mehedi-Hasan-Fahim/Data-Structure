#include<iostream>
using namespace std;

void show(int A[], int size) {
    for (int i = 0; i < size; i++) {
        cout << A[i] << " ";
    }
    cout << endl;
}

void asc_insertion_sort(int A[], int size) {
    int i, j, temp;
    for (i = 1; i < size; i++) {
        temp = A[i];
        j = i - 1;

        while (j >= 0 && A[j] > temp) {
            A[j + 1] = A[j];
            j--;
        }
        A[j + 1] = temp;
    }
}

void dsc_insertion_sort(int A[], int size) {
    int i, j, temp;
    for (i = 1; i < size; i++) {
        temp = A[i];
        j = i - 1;

        while (j >= 0 && A[j] < temp) {
            A[j + 1] = A[j];
            j--;
        }
        A[j + 1] = temp;
    }
}

void asc_selection_sort(int A[], int size) {
    int temp, i, j, index_min;
    for (i = 0; i < size - 1; i++) {
        index_min = i;
        for (j = i + 1; j < size; j++) {
            if (A[j] < A[index_min]) {
                index_min = j;
            }
        }
        if (index_min != i) {
            temp = A[i];
            A[i] = A[index_min];
            A[index_min] = temp;
        }
    }
}

void dsc_selection_sort(int A[], int size) {
    int temp, i, j, index_max;
    for (i = 0; i < size - 1; i++) {
        index_max = i;
        for (j = i + 1; j < size; j++) {
            if (A[j] > A[index_max]) {
                index_max = j;
            }
        }
        if (index_max != i) {
            temp = A[i];
            A[i] = A[index_max];
            A[index_max] = temp;
        }
    }
}

void asc_bubble_sort(int A[], int size) {
    int i, j, temp;
    for (i = 0; i < size - 1; i++) {
        for (j = 0; j < size - i - 1; j++) {
            if (A[j] > A[j + 1]) {
                temp = A[j];
                A[j] = A[j + 1];
                A[j + 1] = temp;
            }
        }
    }
}

void dsc_bubble_sort(int A[], int size) {
    int i, j, temp;
    for (i = 0; i < size - 1; i++) {
        for (j = 0; j < size - i - 1; j++) {
            if (A[j] < A[j + 1]) {
                temp = A[j];
                A[j] = A[j + 1];
                A[j + 1] = temp;
            }
        }
    }
}
