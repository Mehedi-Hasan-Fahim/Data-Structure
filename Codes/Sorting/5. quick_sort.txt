#include<iostream>
using namespace std;

void show(int A[], int size) {
    for (int i = 0; i < size; i++) {
        cout << A[i] << " ";
    }
    cout << endl;
}




void swap(int &a, int &b) {
  int t = a;
  a = b;
  b = t;
}

int asc_partition(int A[],int l,int h){
    int pivot=A[l];
    int i=l;int j=h;
    while(i<j){
        while(A[i]<=pivot){
            i++;
        }
        while(A[j]>pivot){
            j--;
        }
       if(i<j){ 
        swap(A[i],A[j]);
       }
    }
    swap(A[l],A[j]);
    return j;
}


void asc_quick_sort(int A[],int l,int h){
    if(l<h){
        int j=asc_partition(A,l,h);
        asc_quick_sort(A,l,j-1);
        asc_quick_sort(A,j+1,h);
    }
}


int dsc_partition(int A[],int l,int h){
    int pivot=A[l];
    int i=l;int j=h;
    while(i<j){
        while(A[i]>=pivot){
            i++;
        }
        while(A[j]<pivot){
            j--;
        }
       if(i<j){ 
        swap(A[i],A[j]);
       }
    }
    swap(A[l],A[j]);
    return j;
}


void dsc_quick_sort(int A[],int l,int h){
    if(l<h){
        int j=dsc_partition(A,l,h);
        dsc_quick_sort(A,l,j-1);
        dsc_quick_sort(A,j+1,h);
    }
}
