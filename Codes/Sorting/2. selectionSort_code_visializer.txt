#include <iostream>
using namespace std;

void asc_selection_sort(int arr[], int size) {
    int temp, i, j, index_min;
    for (i = 0; i < size - 1; i++) {
        index_min = i;
        for (j = i + 1; j < size; j++) {
            if (arr[j] < arr[index_min]) {
                index_min = j;
            }
        }
        if (index_min != i) {
            swap(arr[i] , arr[index_min]);
        }
    }
}
void show(int arr[], int size) {
    for (int i = 0; i < size; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;
}

int main() {

    int arr[5]={5,12,0,1,8};
    asc_selection_sort(arr, 5);

    return 0;
}
