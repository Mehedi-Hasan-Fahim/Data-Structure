#include<iostream>
using namespace std;

void show(int A[], int size) {
    for (int i = 0; i < size; i++) {
        cout << A[i] << " ";
    }
    cout << endl;
}

void swap(int &a, int &b) {
  int t = a;
  a = b;
  b = t;
}

int asc_partition(int A[],int l,int h){
    int pivot=A[l];
    int i=l;int j=h;
    while(i<j){
        while(A[i]<=pivot){
            i++;
        }
        while(A[j]>pivot){
            j--;
        }
       if(i<j){ 
        swap(A[i],A[j]);
       }
    }
    swap(A[l],A[j]);
    return j;
}


void asc_quick_sort(int A[],int l,int h){
    if(l<h){
        int j=asc_partition(A,l,h);
        asc_quick_sort(A,l,j-1);
        asc_quick_sort(A,j+1,h);
    }
}

int main() {
    int A[10]={12,1,3,21,17,9,11,2,4,10};
    int size =10;
    asc_quick_sort(A,0,size-1);
    show(A, size);


    return 0;
}
