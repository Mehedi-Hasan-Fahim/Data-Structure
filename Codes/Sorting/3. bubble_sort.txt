#include<iostream>
using namespace std;

void show(int A[], int size) {
    for (int i = 0; i < size; i++) {
        cout << A[i] << " ";
    }
    cout << endl;
}


void asc_bubble_sort(int A[], int size) {
    int i, j, temp;
    for (i = 0; i < size - 1; i++) {
        for (j = 0; j < size - i - 1; j++) {
            if (A[j] > A[j + 1]) {
                temp = A[j];
                A[j] = A[j + 1];
                A[j + 1] = temp;
            }
        }
    }
}

void dsc_bubble_sort(int A[], int size) {
    int i, j, temp;
    for (i = 0; i < size - 1; i++) {
        for (j = 0; j < size - i - 1; j++) {
            if (A[j] < A[j + 1]) {
                temp = A[j];
                A[j] = A[j + 1];
                A[j + 1] = temp;
            }
        }
    }
}






int main() {
    int A[50], size;
    char choice;

    cout << "Enter Input Size:";
    cin >> size;

    cout << "Enter Array Elements:" << endl;
    for (int i = 0; i < size; i++) {
        cin >> A[i];
    }

    do {
        cout << "\nMenu:\n";
        cout << "3.Bubble sort\n";
        cout << "6.Exit\n";
        cout << "Enter your choice: ";
        cin >> choice;

        switch (choice) {
            case '3':
                
                    do {
                        cout << "1.ascending  sort\n";
                        cout << "2.descending sort\n";
                        cout << "3.exit\n";
                        cout << "Enter your choice: ";
                        cin >> choice;
                        switch (choice) {
                            case '1':
                                asc_bubble_sort(A, size);
                                show(A, size);
                                break;
                            case '2':
                                dsc_bubble_sort(A, size);
                                show(A, size);
                                break;
                            case '3':
                                break;
                            default:
                                cout << "Invalid choice. Please enter a valid option.\n";
                        }
                    } while (choice != '3');
                
                break;
                
            case '6':
                break;
            default:
                cout << "Invalid choice. Please enter a valid option.\n";
        }
    } while (choice != '6');

    return 0;
}
