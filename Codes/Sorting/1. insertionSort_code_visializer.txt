#include <iostream>
using namespace std;

void asc_insertion_sort(int arr[], int size) {
    int i,j,temp;
    for(i=1;i<size;i++){
        temp = arr[i];
        j=i-1;
        while(j>=0 && arr[j] > temp){
            arr[j+1] = arr[j];
            j--;
        }
        arr[j+1]=temp;
    }
}
void show(int arr[], int size) {
    for (int i = 0; i < size; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;
}

int main() {

    int arr[5]={5,12,0,1,8};
    asc_insertion_sort(arr, 5);
    //show(arr, 5);

    return 0;
}
