#include<iostream>
using namespace std;

void show(int A[], int size) {
    for (int i = 0; i < size; i++) {
        cout << A[i] << " ";
    }
    cout << endl;
}


void asc_merge_sort(int A[],int left,int mid,int right){
    
    
    int n1=mid-left+1;//size_left
    int n2=right-mid;//size_right
    
    int L[n1],R[n2];
    for(int i=0;i<n1;i++){
        L[i]=A[left+i];
    }
    for(int i=0;i<n2;i++){
        R[i]=A[mid+1+i];
    }
    
    int i,j,k;
    i=0;j=0;k=left;
    
    while(i<n1 && j<n2){
        if(L[i]<=R[j]){
           A[k]=L[i];
           i++;
        }else{
            A[k]=R[j];
            j++;
        }
        k++;
    }
    
    while (i < n1) {
        A[k] = L[i];
        i++;
        k++;
    }
    
    while (j < n2) {
        A[k] = R[j];
        j++;
        k++;
    }
    
}
void dsc_merge_sort(int A[],int left,int mid,int right){
    
    
    int n1=mid-left+1;//size_left
    int n2=right-mid;//size_right
    
    int L[n1],R[n2];
    for(int i=0;i<n1;i++){
        L[i]=A[left+i];
    }
    for(int i=0;i<n2;i++){
        R[i]=A[mid+1+i];
    }
    
    int i,j,k;
    i=0;j=0;k=left;
    
    while(i<n1 && j<n2){
        if(L[i]>=R[j]){
           A[k]=L[i];
           i++;
        }else{
            A[k]=R[j];
            j++;
        }
        k++;
    }
    
    while (i < n1) {
        A[k] = L[i];
        i++;
        k++;
    }
    
    while (j < n2) {
        A[k] = R[j];
        j++;
        k++;
    }
    
}


void asc_merge_divide(int A[],int left,int right){
    
    if(left>=right){
        return ;
    }
    int mid =left +(right-left)/2;
    asc_merge_divide(A,left,mid);
    asc_merge_divide(A,mid+1,right);
    asc_merge_sort(A,left,mid,right);
}
void dsc_merge_divide(int A[],int left,int right){
    
    if(left>=right){
        return ;
    }
    int mid =left +(right-left)/2;
    dsc_merge_divide(A,left,mid);
    dsc_merge_divide(A,mid+1,right);
    dsc_merge_sort(A,left,mid,right);
}


void swap(int &a, int &b) {
  int t = a;
  a = b;
  b = t;
}
